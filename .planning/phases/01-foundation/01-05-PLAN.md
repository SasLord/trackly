```
use anyhow::Result;
use tracing_appender::non_blocking::WorkerGuard;
use tracing_subscriber::{EnvFilter, Registry, fmt, prelude::*};
use trackly_infra::{AppConfig, Paths};

pub fn init(paths: &Paths, config: &AppConfig) -> Result<WorkerGuard> {
    std::fs::create_dir_all(paths.logs_dir())?;
    let file_appender = tracing_appender::rolling::daily(paths.logs_dir(), "trackly.log");
    let (non_blocking, guard) = tracing_appender::non_blocking(file_appender);

    let env_filter = EnvFilter::try_from_env("TRACKLY_LOG")
        .unwrap_or_else(|_| EnvFilter::new(format!(
            "{},hyper=warn,tower_http=warn",
            config.logging.level.as_str()
        )));

    if config.logging.format == "json" {
        let file_layer = fmt::layer().json().with_writer(non_blocking).with_ansi(false);
        let stdout_layer = fmt::layer().compact().with_writer(std::io::stderr);
        Registry::default().with(env_filter).with(file_layer).with(stdout_layer).try_init()?;
    } else {
        let file_layer = fmt::layer().compact().with_writer(non_blocking).with_ansi(false);
        let stdout_layer = fmt::layer().compact().with_writer(std::io::stderr);
        Registry::default().with(env_filter).with(file_layer).with(stdout_layer).try_init()?;
    }

    Ok(guard)
}
```

```
#[tokio::test(flavor = "multi_thread")]
async fn health_dto_round_trip_identical() -> anyhow::Result<()> {
    let ctx: AppCtx = test_app_ctx().await?;  // helper below builds a real AppCtx over tempfile DB

    let from_tauri = build_health(&ctx).await;

    let app = router().with_state(ctx.clone());
    let res = app.oneshot(Request::builder().uri("/api/v1/health").body(Body::empty())?).await?;
    let body_bytes = axum::body::to_bytes(res.into_body(), 1024 * 1024).await?;
    let from_axum: HealthDto = serde_json::from_slice(&body_bytes)?;

    assert_eq!(from_tauri, from_axum);
    Ok(())
}
```
