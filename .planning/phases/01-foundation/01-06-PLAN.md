use anyhow::{Context, Result};
use std::path::{Path, PathBuf};
use std::process::Command;
use std::time::Duration;

const PROCMON_URL: &str = "https://download.sysinternals.com/files/ProcessMonitor.zip";

pub fn ensure_procmon_on_path() -> Result<PathBuf> {
    // Try several known names: Procmon.exe, Procmon64.exe (Procmon UI exe; CLI accepts the same flags).
    for name in &["Procmon.exe", "Procmon64.exe"] {
        if let Ok(out) = Command::new("where").arg(name).output() {
            if out.status.success() {
                let first = String::from_utf8_lossy(&out.stdout)
                    .lines().next().map(str::to_string);
                if let Some(p) = first {
                    let path = PathBuf::from(p.trim());
                    if path.exists() {
                        return Ok(path);
                    }
                }
            }
        }
    }

    // Not found — download + extract.
    eprintln!("[procmon-check] ProcMon not on PATH; downloading from {PROCMON_URL}");
    let dl_dir = tempfile::TempDir::new()?;
    let zip_path = dl_dir.path().join("pm.zip");
    let bytes = reqwest::blocking::get(PROCMON_URL)?
        .error_for_status()?
        .bytes()?;
    std::fs::write(&zip_path, &bytes)?;
    eprintln!("[procmon-check] downloaded {} bytes (sha256 logged below for audit)", bytes.len());
    // Log SHA256 for the audit trail (not gated — Sysinternals does not publish stable checksums).
    use sha2::{Sha256, Digest};
    let hash = format!("{:x}", Sha256::digest(&bytes));
    eprintln!("[procmon-check] ProcessMonitor.zip sha256={hash}");

    // Extract.
    let extract_dir = dl_dir.path().join("pm");
    std::fs::create_dir_all(&extract_dir)?;
    let zip_file = std::fs::File::open(&zip_path)?;
    let mut archive = zip::ZipArchive::new(zip_file)?;
    archive.extract(&extract_dir)?;

    // Locate Procmon.exe (or Procmon64.exe on x64).
    for name in &["Procmon64.exe", "Procmon.exe"] {
        let candidate = extract_dir.join(name);
        if candidate.exists() {
            // Persist the extracted dir for the lifetime of the process by leaking the TempDir.
            std::mem::forget(dl_dir);
            return Ok(candidate);
        }
    }
    anyhow::bail!("ProcMon not found in extracted ProcessMonitor.zip")
}

pub fn run_capture(procmon_exe: &Path, target_exe: &Path, sandbox: &Path) -> Result<(PathBuf, PathBuf)> {
    let pml = sandbox.join("trace.pml");
    let csv = sandbox.join("trace.csv");

    // Step A: spawn ProcMon (it runs for /Runtime seconds or until /Terminate).
    let mut procmon = Command::new(procmon_exe)
        .args([
            "/AcceptEula",
            "/Quiet",
            "/Minimized",
            "/Runtime", "30",
            "/BackingFile",
        ])
        .arg(&pml)
        .spawn()
        .context("spawn Procmon for capture")?;

    // Give ProcMon 2 seconds to attach its driver before starting the workload.
    std::thread::sleep(Duration::from_secs(2));

    // Step B: run trackly --self-test and capture exit code.
    let out = Command::new(target_exe)
        .arg("--self-test")
        .output()
        .context("invoke trackly --self-test")?;
    eprintln!("[procmon-check] trackly stderr: {}",
        String::from_utf8_lossy(&out.stderr));
    if !out.status.success() {
        anyhow::bail!("trackly --self-test exited non-zero: {:?}", out.status);
    }

    // Step C: terminate ProcMon cleanly (or wait for /Runtime to expire).
    let _ = Command::new(procmon_exe).arg("/Terminate").status();
    let _ = procmon.wait();

    // Step D: export the PML to CSV (all events; we filter in Rust for clarity).
    let export_status = Command::new(procmon_exe)
        .args([
            "/OpenLog",
        ])
        .arg(&pml)
        .args(["/SaveAs"])
        .arg(&csv)
        .args(["/Quiet", "/AcceptEula"])
        .status()
        .context("export procmon PML to CSV")?;
    if !export_status.success() {
        anyhow::bail!("procmon /SaveAs failed: {:?}", export_status);
    }
    if !csv.exists() {
        anyhow::bail!("procmon CSV not produced at {csv:?}");
    }
    Ok((pml, csv))
}

use anyhow::{Context, Result, bail};
use std::path::Path;

const FORBIDDEN_FRAGMENTS: &[&str] = &[
    "\\AppData\\Local\\",
    "\\AppData\\Roaming\\",
    "\\AppData\\LocalLow\\",
    "\\ProgramData\\",
];

pub fn assert_no_forbidden_writes(csv_path: &Path, sandbox: &Path) -> Result<()> {
    let sandbox_norm = normalize(&sandbox.to_string_lossy());
    let mut rdr = csv::ReaderBuilder::new()
        .has_headers(true)
        .from_path(csv_path)
        .context("open procmon CSV")?;

    let headers = rdr.headers()?.clone();
    let path_col = headers.iter().position(|h| h.eq_ignore_ascii_case("Path"))
        .ok_or_else(|| anyhow::anyhow!("'Path' column not found in CSV headers: {:?}", headers))?;
    let op_col = headers.iter().position(|h| h.eq_ignore_ascii_case("Operation"))
        .ok_or_else(|| anyhow::anyhow!("'Operation' column not found"))?;
    let proc_col = headers.iter().position(|h| h.eq_ignore_ascii_case("Process Name"))
        .ok_or_else(|| anyhow::anyhow!("'Process Name' column not found"))?;

    let mut offenses: Vec<String> = Vec::new();
    for (i, rec) in rdr.records().enumerate() {
        let rec = rec.with_context(|| format!("read row {i}"))?;
        let proc_name = rec.get(proc_col).unwrap_or("");
        if !proc_name.eq_ignore_ascii_case("trackly.exe") { continue; }
        let op = rec.get(op_col).unwrap_or("");
        if !(op == "WriteFile" || op == "CreateFile" || op == "SetEndOfFileInformationFile" || op == "WriteFileGather") {
            continue;
        }
        let path = rec.get(path_col).unwrap_or("");
        let path_norm = normalize(path);

        // Allow: writes inside the sandbox.
        if path_norm.starts_with(&sandbox_norm) { continue; }
        // Allow: Windows temp directory cousins of the sandbox (the sandbox itself lives under %TEMP%).
        if let Some(temp) = std::env::var_os("TEMP") {
            let temp_norm = normalize(&temp.to_string_lossy());
            if path_norm.starts_with(&temp_norm) { continue; }
        }
        // Forbidden: any APPDATA / ProgramData prefix.
        for bad in FORBIDDEN_FRAGMENTS {
            if path_norm.contains(&normalize(bad)) {
                offenses.push(format!("row {i}: {op} -> {path}"));
                break;
            }
        }
    }

    if !offenses.is_empty() {
        bail!(
            "portable-mode leak: {} forbidden write(s) detected:\n  {}",
            offenses.len(),
            offenses.join("\n  "),
        );
    }
    Ok(())
}

fn normalize(s: &str) -> String {
    s.replace('/', "\\").to_ascii_uppercase()
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Sysinternals download (HTTPS GET to `download.sysinternals.com`) → executed binary | Supply-chain: a compromised Sysinternals CDN could ship a malicious `Procmon.exe` that we then execute with admin rights on the CI runner |
| ProcMon driver → kernel | ProcMon loads a kernel filter driver; if the driver is compromised, full kernel compromise of the CI runner |
| `procmon-check.exe` → `trackly.exe` invocation (via `Command::new(...).spawn()`) | Arbitrary command execution scope is bounded by the `trackly.exe` path we control |
| CSV file produced by ProcMon → `csv` crate parser | Malformed CSV could trigger parser bugs; `csv 1.3` is battle-tested |
| Cyrillic path on Windows filesystem APIs → file I/O | Encoding bugs in CreateFileW vs CreateFileA could cause silent failures (Pitfall #3) |
