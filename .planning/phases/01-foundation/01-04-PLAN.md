```
use serde::{Serialize, Serializer};

#[derive(Debug, thiserror::Error)]
pub enum AppError {
    #[error("{entity} {id} not found")]
    NotFound { entity: &'static str, id: i64 },

    #[error("conflict: {reason}")]
    Conflict { reason: String },

    #[error("optimistic lock mismatch on {entity} {id}: expected v{expected}, found v{actual}")]
    OptimisticLockMismatch { entity: &'static str, id: i64, expected: i64, actual: i64 },

    #[error("write queue busy (5s timeout)")]
    WriteQueueBusy,

    #[error("database from newer version: binary={binary}, file={file}")]
    DatabaseFromNewerVersion { binary: u32, file: u32 },

    #[error("validation: {field}: {message}")]
    Validation { field: String, message: String },

    #[error("unauthorized")]
    Unauthorized,

    #[error("forbidden")]
    Forbidden,

    #[error("internal: {source_chain}")]
    Internal { source_chain: String },
}

impl AppError {
    pub fn code(&self) -> &'static str {
        match self {
            Self::NotFound { .. } => "NOT_FOUND",
            Self::Conflict { .. } => "CONFLICT",
            Self::OptimisticLockMismatch { .. } => "OPTIMISTIC_LOCK_MISMATCH",
            Self::WriteQueueBusy => "WRITE_QUEUE_BUSY",
            Self::DatabaseFromNewerVersion { .. } => "DATABASE_FROM_NEWER_VERSION",
            Self::Validation { .. } => "VALIDATION",
            Self::Unauthorized => "UNAUTHORIZED",
            Self::Forbidden => "FORBIDDEN",
            Self::Internal { .. } => "INTERNAL",
        }
    }
}

impl Serialize for AppError {
    fn serialize<S: Serializer>(&self, ser: S) -> Result<S::Ok, S::Error> {
        use serde::ser::SerializeStruct;
        let mut s = ser.serialize_struct("AppError", 3)?;
        s.serialize_field("code", self.code())?;
        s.serialize_field("message", &self.to_string())?;
        // details: a serde_json::Value per variant; small helper below
        s.serialize_field("details", &self.details_value())?;
        s.end()
    }
}

// details_value returns a serde_json::Value with the variant-specific fields
// (e.g., NotFound -> {"entity": "...", "id": 123}).
```

`crates/trackly-infra/src/db/writer_worker.rs` (Pattern 1 from RESEARCH, closure form per Open Question #1):
```
use rusqlite::Connection;
use std::time::Duration;
use tokio::sync::{mpsc, oneshot};
use trackly_core::error::AppError;

type BoxedJob = Box<dyn FnOnce(&mut Connection) + Send + 'static>;

pub struct WriterHandle { tx: mpsc::Sender<BoxedJob> }

impl WriterHandle {
    pub fn spawn(conn: Connection) -> Self { Self::spawn_with_capacity(conn, 256) }

    pub fn spawn_with_capacity(mut conn: Connection, capacity: usize) -> Self {
        let (tx, mut rx) = mpsc::channel::<BoxedJob>(capacity);
        tokio::task::spawn_blocking(move || {
            while let Some(job) = rx.blocking_recv() {
                job(&mut conn);
            }
        });
        Self { tx }
    }

    pub async fn execute<F, R>(&self, op: F) -> Result<R, AppError>
    where
        F: FnOnce(&mut Connection) -> Result<R, AppError> + Send + 'static,
        R: Send + 'static,
    {
        let (reply_tx, reply_rx) = oneshot::channel();
        let job: BoxedJob = Box::new(move |conn| { let _ = reply_tx.send(op(conn)); });
        self.tx.send_timeout(job, Duration::from_secs(5)).await
            .map_err(|_| AppError::WriteQueueBusy)?;
        reply_rx.await.map_err(|_| AppError::Internal {
            source_chain: "writer worker dropped reply channel".into(),
        })?
    }
}
```

`crates/trackly-infra/src/db/pools.rs`:
```
use rusqlite::{Connection, OpenFlags};
use std::path::Path;
use std::sync::Mutex;
use trackly_core::error::AppError;

pub struct ReaderPool { conns: Mutex<Vec<Connection>> }

impl ReaderPool {
    pub fn new(db_path: &Path, size: usize) -> Result<Self, AppError> {
        let mut conns = Vec::with_capacity(size);
        for _ in 0..size {
            let conn = Connection::open_with_flags(
                db_path,
                OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_NO_MUTEX,
            ).map_err(AppError::from)?;
            crate::db::pragmas::apply_reader_pragmas(&conn)?;
            conns.push(conn);
        }
        Ok(Self { conns: Mutex::new(conns) })
    }

    pub fn acquire(&self) -> ReaderHandle<'_> {
        let conn = self.conns.lock().unwrap().pop()
            .expect("ReaderPool exhausted — bump size or audit long-running reads");
        ReaderHandle { pool: self, conn: Some(conn) }
    }
}

pub struct ReaderHandle<'a> {
    pool: &'a ReaderPool,
    conn: Option<Connection>,
}
impl<'a> std::ops::Deref for ReaderHandle<'a> {
    type Target = Connection;
    fn deref(&self) -> &Connection { self.conn.as_ref().unwrap() }
}
impl<'a> Drop for ReaderHandle<'a> {
    fn drop(&mut self) {
        if let Some(c) = self.conn.take() {
            self.pool.conns.lock().unwrap().push(c);
        }
    }
}
```
(Simpler than `deadpool`; sufficient for LAN scale. If contention surfaces later, swap to `deadpool` without changing the public surface.)

`AppCtx::build(paths: Paths, config: AppConfig, log_guard: WorkerGuard) -> anyhow::Result<Self>` lifecycle (the probe-read pattern is mandatory — see W4 in the revision context):

```
impl AppCtx {
    pub async fn build(paths: Paths, config: AppConfig, log_guard: WorkerGuard) -> anyhow::Result<Self> {
        use rusqlite::{Connection, OpenFlags};

        // Step 6a: resolve db_path (config override or paths default)
        let db_path = if !config.paths.db_path.is_empty() {
            std::path::PathBuf::from(&config.paths.db_path)
        } else {
            paths.db_path().to_path_buf()
        };

        // Step 6b: PROBE-READ user_version with a READ-ONLY conn.
        // SQLITE_OPEN_READ_ONLY guarantees SQLite does NOT mutate the main file or
        // touch the WAL — so a rejected downgrade leaves the file byte-identical.
        // SQLITE_OPEN_URI keeps URI-style paths working if config.paths.db_path is one.
        // If the DB file does not yet exist, skip the probe and proceed to writer-open
        // (which will create the file); first-run cannot be a downgrade.
        if db_path.exists() {
            let known: u32 = trackly_infra::db::migrations::max_known_version();
            let probe = Connection::open_with_flags(
                &db_path,
                OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_URI,
            )?;
            let on_disk: u32 = probe
                .pragma_query_value(None, "user_version", |row| row.get::<_, i64>(0))? as u32;
            drop(probe);  // explicit: release the read-only handle before opening writer
            if on_disk > known {
                return Err(trackly_core::error::AppError::DatabaseFromNewerVersion {
                    binary: known, file: on_disk,
                }.into());
            }
        }

        // Step 7: open WRITER conn (creates file if missing; WAL initialization on first open
        // legitimately mutates the file — that is acceptable now that the downgrade check
        // has already gated this branch).
        let mut writer_conn = Connection::open(&db_path)?;
        trackly_infra::db::pragmas::apply_writer_pragmas(&writer_conn)?;

        // Step 8: run migrations
        let report = trackly_infra::db::migrations::run(&mut writer_conn)?;
        let schema_version = report.schema_version;

        // Step 9: hand writer conn to spawn_blocking worker
        let writer = Arc::new(WriterHandle::spawn(writer_conn));

        // Step 10: open reader pool
        let readers = Arc::new(ReaderPool::new(&db_path, 4)?);

        Ok(Self {
            writer, readers,
            paths: Arc::new(paths), config: Arc::new(config),
            clock: Arc::new(trackly_infra::clock_impl::SystemClock),
            shutdown: CancellationToken::new(),
            _log_guard: Arc::new(log_guard),
            schema_version,
        })
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| `AppCtx` consumers (Tauri/axum handlers) → DB | Every write MUST go through `WriterHandle::execute`; bypassing it via direct `rusqlite::Connection` reintroduces SQLITE_BUSY |
| User-controlled DB file (could be from older version) → `AppCtx::build` | An attacker swapping in a maliciously-crafted DB file could try to corrupt state; downgrade-protection mitigates the "newer version" case |
| Secrets in process memory → log/debug output | Without `Secret<T>`, accidental `format!("{:?}", config)` could leak passwords (relevant Phase 5+) |
| Future contributor → AppError variants | Adding a variant without updating `code()` returns "INTERNAL" by accident, breaking frontend error matching |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | DB file from a newer schema version mounted under the binary; binary tries to migrate → corruption | mitigate | `AppCtx::build` PROBE-READS `PRAGMA user_version` via a `SQLITE_OPEN_READ_ONLY` conn BEFORE opening the writer; if file > binary, return `DatabaseFromNewerVersion` and abort. Read-only open does not touch WAL or main file, so file is byte-identical. Test `downgrade_protection.rs` asserts SHA256 equality of `.db` + `.db-wal`. |
| T-04-02 | Denial of Service | 1000 concurrent write requests overwhelm the writer queue, causing memory exhaustion | mitigate | mpsc channel is BOUNDED at 256; the 257th caller blocks for up to 5s then receives `AppError::WriteQueueBusy`. Phase 1 lacks load metrics; Phase 5+ should log queue depth periodically. Recommended for SUMMARY |
| T-04-03 | Information Disclosure | `Debug` impl on a struct containing `Secret<T>` leaks the secret | mitigate | `Secret<T>::Debug = "***"` (Plan 04); the type cannot derive `Serialize`/`Deserialize` (compile-time check via `static_assertions`); the `Drop` impl zeroizes the inner value so post-drop memory inspection finds zeros |
| T-04-04 | Tampering | Writer worker panics inside the closure, killing the worker but leaving the channel open → subsequent `execute` calls hang indefinitely | mitigate (partial) | The worker loop does NOT catch panics; if a closure panics, the worker task aborts. | T-04-05 | Denial of Service | Reader pool exhaustion (5th reader requested when pool is 4) panics the calling task | accept (with note) | Phase 1 ships a simple `Vec<Connection>` pool that panics on exhaustion.
