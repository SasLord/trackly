```rust
/// Стабильный сигнатурный ключ FK-нарушения, ИГНОРИРУЮЩИЙ `rowid` (WR-01):
/// perестройка таблицы (CREATE TABLE new -> INSERT SELECT -> DROP -> RENAME,
/// норма проекта — V042/V043/V044) переназначает rowid, поэтому сравнение по
/// сырому кортежу либо ложно репортит перестроенное legacy-нарушение как
/// новое, либо маскирует генуинно новое нарушение на переиспользованном
/// rowid. Сравнение по МНОЖЕСТВУ (table, parent, fkid) с УЧЁТОМ КРАТНОСТИ
/// (BTreeMap -> count) устойчиво к обоим случаям.
type FkSignature = (String, String, i64);

fn signature_counts(violations: &BTreeSet<FkViolation>) -> BTreeMap<FkSignature, usize> {
    let mut out = BTreeMap::new();
    for (table, _rowid, parent, fkid) in violations {
        *out.entry((table.clone(), parent.clone(), *fkid)).or_insert(0) += 1;
    }
    out
}

/// Rowid-стабильная замена `after.difference(&baseline)`: элементы `after`,
/// чья (table, parent, fkid)-сигнатура встречается ЧАЩЕ, чем в `baseline`.
fn violations_with_grown_signature<'a>(
    after: &'a BTreeSet<FkViolation>,
    baseline: &BTreeSet<FkViolation>,
) -> Vec<&'a FkViolation> {
    let after_counts = signature_counts(after);
    let baseline_counts = signature_counts(baseline);
    let mut remaining: BTreeMap<FkSignature, usize> = BTreeMap::new();
    for (sig, count) in &after_counts {
        let base = baseline_counts.get(sig).copied().unwrap_or(0);
        if *count > base {
            remaining.insert(sig.clone(), count - base);
        }
    }
    if remaining.is_empty() {
        return Vec::new();
    }
    let mut out = Vec::new();
    for v @ (table, _rowid, parent, fkid) in after {
        let sig = (table.clone(), parent.clone(), *fkid);
        if let Some(left) = remaining.get_mut(&sig) {
            if *left > 0 {
                out.push(v);
                *left -= 1;
            }
        }
    }
    out
}
```

— всё, что идёт ПОСЛЕ этой строки (sample-building, error formatting) не меняется.
