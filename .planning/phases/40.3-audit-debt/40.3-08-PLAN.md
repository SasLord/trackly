```js
const DIGIT_PREFIX_REGEX_LITERAL_RE = /\/\^\(?\\?(?:d|\[0-9\])/g;
// Матчит: /^\d, /^(\d, /^[0-9] — то есть литерал, инлайновый ИЛИ вынесенный в const,
// независимо от группы/класса/бэкслеша.
const NUMBER_RAW_RE = /\bnumber_raw\b/g;
const ALLOWED_NUMBER_RAW_FILES = new Set(['src/features/acts/ActFormBody.svelte']);

function scanSource(relPath, src, violations) {
  DIGIT_PREFIX_REGEX_LITERAL_RE.lastIndex = 0;
  let m;
  while ((m = DIGIT_PREFIX_REGEX_LITERAL_RE.exec(src)) !== null) {
    violations.push(`${relPath}:${lineAt(src, m.index)}: digit-prefix regex-литерал ...`);
  }
  if (!ALLOWED_NUMBER_RAW_FILES.has(relPath)) {
    NUMBER_RAW_RE.lastIndex = 0;
    let m2;
    while ((m2 = NUMBER_RAW_RE.exec(src)) !== null) {
      violations.push(`${relPath}:${lineAt(src, m2.index)}: number_raw вне whitelist ...`);
    }
  }
}

function checkFile(root, relPath, violations) {
  scanSource(relPath, fs.readFileSync(path.join(root, relPath), 'utf8'), violations);
}

function runSelfTest() {
  const fixtures = [
    { name: 'literal /^\\d+/',        relPath: 'fixture.ts',                             src: "n.replace(/^\\d+/, '');",                     expectViolations: 1 },
    { name: 'group /^(\\d+)/',        relPath: 'fixture.ts',                             src: "n.replace(/^(\\d+)/, '');",                   expectViolations: 1 },
    { name: 'char class /^[0-9]+/',   relPath: 'fixture.ts',                             src: "n.replace(/^[0-9]+/, '');",                   expectViolations: 1 },
    { name: 'extracted const',        relPath: 'fixture.ts',                             src: "const RE = /^\\d+/; n.replace(RE, '');",       expectViolations: 1 },
    { name: 'number_raw concat',      relPath: 'src/features/acts/FakeFile.svelte',      src: '`${act.number_raw}в${sub}`',                   expectViolations: 1 },
    { name: 'clean code',             relPath: 'fixture.ts',                             src: "const x = n.trim();",                          expectViolations: 0 },
    { name: 'whitelisted number_raw', relPath: 'src/features/acts/ActFormBody.svelte',   src: 'initialAct!.number_raw',                       expectViolations: 0 },
  ];
  let failed = 0;
  for (const f of fixtures) {
    const v = [];
    scanSource(f.relPath, f.src, v);
    const ok = v.length === f.expectViolations;
    console.error(`${TAG} [selftest] ${ok ? 'PASS' : 'FAIL'} — ${f.name} (expected ${f.expectViolations}, got ${v.length})`);
    if (!ok) failed++;
  }
  return failed === 0;
}
```
`main()` получает новую ветку: если `process.argv` содержит `--selftest`, вызвать
`runSelfTest()` и `process.exit(ok ? 0 : 1)`, НЕ сканируя реальный репозиторий в этом режиме.
