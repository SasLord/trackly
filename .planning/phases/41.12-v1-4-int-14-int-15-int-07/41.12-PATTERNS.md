impl StatusLookup {
    async fn load(readers: Arc<ReaderPool>) -> Result<Self, AppError> {
        let rows = tokio::task::spawn_blocking(move || {
            let conn = readers.acquire();
            let mut stmt = conn
                .prepare("SELECT id, name, COALESCE(code, '') FROM device_statuses ORDER BY id")
                .map_err(map_rusqlite)?;
            let rows = stmt
                .query_map([], |r| Ok((r.get(0)?, r.get(1)?, r.get(2)?)))
                .map_err(map_rusqlite)?
                .collect::<Result<Vec<_>, _>>()
                .map_err(map_rusqlite)?;
            Ok::<_, AppError>(rows)
        })
        .await
        .map_err(|e| AppError::Internal { source_chain: format!("spawn_blocking status lookup: {e}") })??;
        Self::from_rows(rows)
    }

    fn id_for_label(&self, label: Option<&str>) -> i64 {
        let Some(label) = label.map(str::trim).filter(|l| !l.is_empty()) else {
            return self.default_id;
        };
        let wanted = label.to_lowercase();     // lower() SQLite кириллицу не сворачивает
        if let Some((id, _, _)) = self.rows.iter().find(|(_, name, _)| name.to_lowercase() == wanted) {
            return *id;
        }
        if wanted == LEGACY_REPAIR_LABEL_LOWER { /* псевдоним */ }
        self.default_id
    }

    fn name_for_id(&self, id: i64) -> String {
        self.rows.iter().find(|(row_id, _, _)| *row_id == id)
            .map(|(_, name, _)| name.clone())
            .unwrap_or_else(|| format!("Статус {id}"))   // нейтральная подпись для неизвестного id
    }
}
