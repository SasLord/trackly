    Добавить публичную функцию:
    ```rust
    impl ActService {
        pub async fn suggest_person(
            &self,
            field: SuggestPersonField,  // enum {Giver, Receiver}
            prefix: &str,
            limit: usize,
        ) -> Result<Vec<String>, AppError> {
            if prefix.len() > 100 {
                return Err(AppError::Validation { field: "prefix".into(), message: "prefix слишком длинный".into() });
            }
            let escaped = escape_like(prefix);
            let column = match field { SuggestPersonField::Giver => "giver_name", SuggestPersonField::Receiver => "receiver_name" };
            // ReaderPool acquire → query_map → collect
            // SQL — format!() для column name OK (whitelisted via enum); prefix через ?1 prepared.
            let sql = format!(
                "SELECT {col} AS name, COUNT(*) AS freq
                   FROM acts
                  WHERE {col} LIKE ?1 || '%' ESCAPE '\\'
                    AND deleted_at_utc IS NULL
                  GROUP BY {col}
                  ORDER BY freq DESC, {col} ASC
                  LIMIT ?2",
                col = column
            );
            // ...query, collect names, return.
        }
    }
    ```
