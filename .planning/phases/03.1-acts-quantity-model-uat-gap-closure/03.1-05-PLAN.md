    ```rust
    fn truncate_to_width(text: &str, font_size: f32, max_width: f32) -> String {
        let avg_glyph_w = 0.5 * font_size;  // approximation для DejaVu Sans
        let max_chars = (max_width / avg_glyph_w).floor() as usize;
        // B-3 INVARIANT: input strings ≤ max_chars MUST pass through byte-identically (early return).
        // The pdf_determinism::fixture_act_42_renders_to_known_hash test pins SHA256 of fixture
        // act_42 PDF. Fixture rows have short cell text; truncate must be a no-op для них,
        // или SHA256 drift сломает existing tests.
        if text.chars().count() <= max_chars {
            return text.to_string();  // BYTE-IDENTICAL early return — no allocation differences,
                                       // no text shaping side effects, no whitespace adjustments.
        }
        if max_chars <= 1 {
            return "…".to_string();
        }
        let truncated: String = text.chars().take(max_chars - 1).collect();
        format!("{}…", truncated)
    }
    ```
