use std::time::Duration;
use async_trait::async_trait;
use ldap3::{ldap_escape, LdapConnAsync, LdapConnSettings, Scope, SearchEntry};
use trackly_core::error::AppError;
use trackly_core::ports::ad::{AdClient, AuthOutcome};
use trackly_core::primitives::secret::Secret;
use crate::config::AdConfig;

const CONN_TIMEOUT: Duration = Duration::from_secs(5);

pub struct RealAdClient { cfg: AdConfig }

impl RealAdClient {
    fn normalize_bind_name(&self, login: &str) -> String {
        if login.contains('@') || login.contains('\\') { login.to_string() }
        else { format!("{login}@{}", self.cfg.domain) }
    }
}

fn build_user_search_filter(login: &str) -> String {
    let safe_login = ldap_escape(login);
    format!("(|(sAMAccountName={safe_login})(userPrincipalName={safe_login}))")
}

// authenticate(): connect (Err -> Unreachable) -> ldap3::drive!(conn) -> simple_bind
// (Err -> Unreachable; rc!=0 -> BadCreds) -> search(base_dn, Subtree, filter,
// [name_attr, "cn"]) -> displayName -> cn -> login fallback chain -> unbind -> Ok(Outcome)

fn build_group_membership_filter(sam_account_name: &str, group_dn: &str) -> String {
