/// NOTE (full-parity follow-up): `display_name` currently falls back to the SAM login
/// because SSO has no bind to search from. A service-account displayName lookup (as in
/// the adwebapp reference) is deferred to the AD-SSO milestone.
pub async fn sso_login(&self, ad_username: &str, display_name: &str) -> Result<UserDto, AppError> {
    if !self.ad_enabled().await? { return Err(AppError::Unauthorized); }
    self.on_ad_bind_success(ad_username, display_name).await
}

async fn on_ad_bind_success(&self, login: &str, display_name: &str) -> Result<UserDto, AppError> {
    match self.find_user_any_state(login).await? {
        Some(found) if found.is_active && !found.deleted => self.get_by_login(login).await,
        Some(pending) if !pending.is_active && !pending.deleted && pending.has_open_register_request => {
            self.reuse_or_create_pending_registration(pending.id, login, display_name).await
        }
        Some(blocked_or_deleted) => self.report_blocked_access(blocked_or_deleted.id).await,
        None => {
            if self.ad_auto_accept().await? {
                self.auto_register_ad_user(login, display_name).await
            } else {
                self.create_pending_registration(login, display_name).await
            }
        }
    }
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| `AdDirectory::resolve` result -> RBAC role assignment | The `role_hint` returned from directory resolution directly determines the role a NEWLY auto-registered/pending SSO user receives — the single highest-stakes trust boundary in this phase |
| SSO enrichment failure -> login availability | A directory-lookup failure must degrade gracefully (bare login, no role), never block the underlying Kerberos-authenticated login itself — availability vs. privilege are separate concerns |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-31-03a | Elevation of Privilege | `sso_login`'s `Err(DirectoryError::*)` branches | mitigate | Every non-`Ok` outcome of `directory.resolve` maps to `role: None` (never a default-to-elevated fallback); only `Ok(DirectoryResult { role: Some(_), .. })` can elevate, and only reaches an INSERT via the two identified call sites |
| T-31-03b | Denial of Service | Directory-lookup failure blocking SSO login | mitigate | `Err(DirectoryError::Unreachable/ServiceBindFailed/NotConfigured)` degrades `sso_login` to the bare-login/no-role path instead of propagating `ServiceUnavailable` — SSO availability is decoupled from the OPTIONAL enrichment feature, per Pitfall 5 |
| T-31-03c | Repudiation | Silent directory-lookup failures | mitigate | `Unreachable`/`ServiceBindFailed` emit a `tracing::warn!` with the login and error, distinguishable from the silent `NotConfigured` steady state — an admin can tell "AD is down" from "AD directory enrichment isn't configured" |
| T-31-SC | Tampering | npm/pip/cargo installs | accept | No new external packages introduced by this plan |
</threat_model>
