   - insert_model_in_tx, update_model_in_tx

8. get_history(conn, cartridge_id: i64) -> Result<Vec<AuditEntryRow>, AppError>: SELECT entity_type, entity_id, action, user_id, before_json, after_json, payload_json, created_at_utc FROM audit_log WHERE entity_type='cartridge' AND entity_id=?1 AND action NOT IN ('list','get') ORDER BY created_at_utc DESC.
