```rust
pub fn soft_delete_in_tx(
    &self, tx: &Transaction<'_>, id: i64, version: i64, now_utc: i64,
) -> Result<(), AppError> {
    let affected = tx.execute(
        "UPDATE acts SET deleted_at_utc = ?1, version = version + 1, updated_at_utc = ?1 \
         WHERE id = ?2 AND version = ?3 AND deleted_at_utc IS NULL",
        params![now_utc, id, version],
    ).map_err(map_rusqlite)?;
    if affected == 0 {
        let actual: Option<i64> = tx
            .query_row("SELECT version FROM acts WHERE id = ?1", params![id], |r| r.get(0))
            .optional().map_err(map_rusqlite)?;
        return match actual {
            None => Err(AppError::NotFound { entity: "act", id }),
            Some(actual) => Err(AppError::OptimisticLockMismatch { entity: "act", id, expected: version, actual }),
        };
    }
    Ok(())
}
```
