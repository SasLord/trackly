<!-- New PrinterModel-basis SQL — anti-fan-out via correlated EXISTS subquery, mirrors the
     proven pattern in `compatible_model_aggregates` (same file, ~line 353) which already
     documents WHY a direct JOIN through cartridge_model_compatibility is wrong: multiple
     compatibility rows referencing the same (model, printer_name) pair would multiply-count
     cartridges via join fan-out. This query counts each cartridge exactly once regardless of
     how many compatibility rows exist. -->
```sql
SELECT display_name, cnt
  FROM (
    SELECT pg.display_name AS display_name,
           (
             SELECT COUNT(*)
               FROM cartridges c
               JOIN cartridge_models m ON m.id = c.model_id AND m.deleted_at_utc IS NULL
              WHERE c.status_id = 1 AND c.state_id = 1 AND c.deleted_at_utc IS NULL
                AND EXISTS (
                      SELECT 1 FROM cartridge_model_compatibility cmc2
                       WHERE cmc2.cartridge_model_id = m.id
                         AND LOWER(TRIM(cmc2.printer_name)) = pg.norm_name
                    )
           ) AS cnt
      FROM (
            SELECT LOWER(TRIM(printer_name)) AS norm_name,
                   MIN(printer_name) AS display_name
              FROM cartridge_model_compatibility
             GROUP BY LOWER(TRIM(printer_name))
           ) pg
  )
 WHERE cnt < ?1
 ORDER BY cnt ASC, display_name ASC
```
